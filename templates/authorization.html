DCL Commands Used in this Project
1. User Creation & Granting Privileges:
-- 1. Create User
CREATE USER IF NOT EXISTS 'grocery_viewer'@'localhost'
IDENTIFIED BY 'Grocery@123';

-- 2. Grant SELECT privileges on views & tables
GRANT SELECT ON GroceryShopDB.Product_View TO 'grocery_viewer'@'localhost';
GRANT SELECT ON GroceryShopDB.PRODUCT TO 'grocery_viewer'@'localhost';
GRANT SELECT ON GroceryShopDB.Bill_Summary_View TO 'grocery_viewer'@'localhost';

FLUSH PRIVILEGES;
2. Revoking Privileges & Inspection:
-- 3. Show active grants
SHOW GRANTS FOR 'grocery_viewer'@'localhost';

-- 4. Revoke Privilege (if needed)
REVOKE INSERT, UPDATE, DELETE 
ON GroceryShopDB.* 
FROM 'grocery_viewer'@'localhost';
